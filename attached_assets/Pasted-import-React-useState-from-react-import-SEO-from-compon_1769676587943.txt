function compactObject<T extends Record<string, any>>(obj: T) {
  const out: Record<string, any> = {};
  Object.entries(obj).forEach(([k, v]) => {
    if (v !== undefined && v !== null) {
      const s = typeof v === "string" ? v.trim() : v;
      if (s !== "") out[k] = s;
    }
  });
  return out;
}

function buildSubject(form: any) {
  const parts = [
    "Website Booking Inquiry",
    form.name ? `— ${form.name}` : "",
    form.checkIn && form.checkOut
      ? ` (${form.checkIn} → ${form.checkOut})`
      : "",
  ].filter(Boolean);
  return parts.join(" ");
}

function buildBody(form: any) {
  const payload = compactObject(form);

  const summaryLines = [
    `Name: ${form.name || "—"}`,
    `Email: ${form.email || "—"}`,
    `Phone: ${form.phone || "—"}`,
    `Inquiry Type: ${form.inquiryType || "—"}`,
    `Check-in (YYYY-MM-DD): ${form.checkIn || "—"}`,
    `Check-out (YYYY-MM-DD): ${form.checkOut || "—"}`,
    `Guests: ${form.guests || "—"}`,
    // `Property Interest: ${form.propertyInterest || "—"}`,
  ].join("\n");

  // JSON → pretty string → strip curly braces → remove empty lines
  const prettyJsonRaw = JSON.stringify(payload, null, 2);
  const prettyJsonNoBraces = prettyJsonRaw.replace(/[{}]/g, "");
  const prettyJsonTidy = prettyJsonNoBraces
    .split("\n")
    .filter((line) => line.trim() !== "")
    .join("\n");

  return (
    "Inquiry Summary:\n" + summaryLines + "\n\nMessage: " + form.message ||
    "-" + "\n"
  );
}

function createMailto(form: any) {
  const subject = encodeURIComponent(buildSubject(form));
  const body = encodeURIComponent(buildBody(form));
  return `mailto:${encodeURIComponent(
    SUPPORT_EMAIL,
  )}?subject=${subject}&body=${body}`;
}
